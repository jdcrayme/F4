cmake_minimum_required(VERSION 3.20)
project(F4 LANGUAGES C CXX)

# Must be called at the TOP level (before any add_subdirectory) so that
# `ctest` run from the build root discovers every library's tests. Each
# subdirectory's tests/CMakeLists.txt already calls enable_testing() locally
# (which is a no-op once it's been called at the root), but without this
# line the top-level `ctest` invocation reports "No tests were found!!!"
# even though the test executables are built and registered with CTest in
# their subdirectories.
enable_testing()

# Shared module path so include(cmake/f4_*.cmake) works from any subdirectory.
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# Shared warning flags — applied to every f4-* target via target_link_libraries
# (see each library's CMakeLists.txt). See cmake/f4_warnings.cmake for the
# flag selection rationale.
include(f4_warnings)

# Shared dependencies (GoogleTest). Fetched once at the root so each module's
# tests/CMakeLists.txt doesn't repeat the FetchContent block. See
# cmake/f4_deps.cmake.
include(f4_deps)

# Path to the JSON fixtures generated by f4-convert at build time.
#
# Defined here, BEFORE any add_subdirectory(), so that f4-data/tests can see
# it on the FIRST configure pass. f4-data is configured before f4-convert
# (see ordering below), so a cache variable set inside f4-convert/tests would
# be empty when f4-data reads it on a fresh configure, causing the 5
# fixture-consuming tests to silently SKIP. Defining it at the root fixes
# that without reordering the libraries (f4-data still links only f4-math).
set(F4_GENERATED_FIXTURES_DIR "${CMAKE_BINARY_DIR}/generated_fixtures" CACHE PATH
    "Path to JSON fixtures generated by f4-convert at build time")

add_subdirectory(f4-units)
add_subdirectory(f4-math)
add_subdirectory(f4-json)         # minimal dependency-free JSON reader/writer
add_subdirectory(f4-xml)          # vendored pugixml XML parser (SVG symbol import; future BMS XML data tables)
add_subdirectory(f4-assets)      # runtime asset root, manifest reader, capability/staleness API (see Docs/ASSET_PIPELINE_SPEC.md)
add_subdirectory(f4-import)      # importer side: doctor + f4import CLI (see Docs/ASSET_PIPELINE_SPEC.md)
add_subdirectory(f4-gltf)        # runtime glTF 2.0 loader (Stage 3, replaces f4-models in the runtime link closure)
add_subdirectory(f4-geo)          # strong-typed coordinate frames & geodesy
add_subdirectory(f4-state-machine)
add_subdirectory(f4-entities)     # component-based entity system (uses f4-geo)
add_subdirectory(f4-messaging)    # type-safe message bus with cross-thread queuing
add_subdirectory(f4-data)
add_subdirectory(f4-io)           # shared binary I/O primitives (Cursor + read_file) — consumed by f4-world-convert and f4-terrain
add_subdirectory(f4-weapons)      # weapons & effects core (missile flyout, gun, damage, stores) — see Docs/COMBAT_CHAIN_PLAN.md
add_subdirectory(f4-sensors)      # sensor model (radar detection, track files, RWR) — see Docs/COMBAT_CHAIN_PLAN.md
add_subdirectory(f4-convert)
add_subdirectory(f4-install)      # locates Falcon 4.0 install layout (FALCON4.ct, terrdata/, campaign/)
add_subdirectory(f4-world-convert) # .cam campaign archive -> JSON (keystone)
add_subdirectory(f4-world)         # typed WorldState -> populates f4-entities
add_subdirectory(f4-terrain)       # theater terrain (elevation/palette/overlay)
add_subdirectory(f4-terrain-convert) # THEATER.* -> korea.terrain.json (CLI)
add_subdirectory(f4-lzss)           # LZSS decompression (TEX blobs, CAM archives)
add_subdirectory(f4-models)        # KoreaObj.HDR/LOD 3D model database parser
option(F4_BUILD_RENDERER "Build the f4-renderer Raylib library (requires X11 + OpenGL dev headers)" ON)
if(F4_BUILD_RENDERER)
    add_subdirectory(f4-renderer)       # Shared Raylib rendering library (orbit camera, lit shader, mesh builder, texture cache, symbols)
endif()
option(F4_BUILD_MODEL_VIEWER "Build the 3D model viewer (requires X11 + OpenGL dev headers)" ON)
if(F4_BUILD_MODEL_VIEWER)
    add_subdirectory(f4-models-viewer)  # Raylib + Dear ImGui 3D model viewer
endif()
# NOTE: f4-world-viewer moved BELOW f4-simulation (the V-CAMP tranche):
# the viewer's live campaign session links f4-simulation and reads the
# campaign's F4_MISSION_PROFILES_JSON cache var at configure time — the
# same ordering rule that put f4-campaign before f4-simulation (the
# f4-campaign comment below). Raylib/ImGui targets it also needs
# (f4-renderer, f4-models) are all added above this point.
add_subdirectory(f4-flight-api)      # Flight model API interfaces (PilotInput, IAircraftState, IPilotInputSink)
add_subdirectory(f4-flight-model)
add_subdirectory(f4-recorder)        # Flight recording, replay, and LLM-friendly data export
add_subdirectory(f4-ai)             # AI brain: SensorFusion + tactic modules (composed)
# f4-campaign BEFORE f4-simulation: the QC harness's synthetic tasking
# ladder consumes the campaign's generated MissionProfiles.json (cache
# var + fixture target must exist when f4-simulation is configured).
add_subdirectory(f4-campaign)       # Headless dynamic campaign (mission profiles, tasking tick, attrition) — see Docs/NEXT_PHASE_PLAN.md §B
add_subdirectory(f4-simulation)     # Orchestration: EntityWorld + MessageBus + ModelDatabase + tick loop (no rendering)
# The viewer AFTER f4-campaign + f4-simulation (see the note above its
# former location): the CampaignSession composes the campaign loop with
# the Simulation, and its tests consume both fixture surfaces.
option(F4_BUILD_VIEWER "Build the Raylib + Dear ImGui interactive viewer (requires X11 + OpenGL dev headers)" ON)
if(F4_BUILD_VIEWER)
    add_subdirectory(f4-world-viewer)  # Raylib + Dear ImGui interactive viewer
endif()

# ── Configure the scenario JSON templates at build time ────────────────────
# This block runs regardless of whether f4-scenario-player is built (the
# scenarios are pure JSON consumed by f4-simulation tests + the player).
# Substitutes @F4_SOURCE_DIR@, @F4_BINARY_DIR@, @F4_DIGI_CLASS_TABLE@ so the
# scenarios can reference KoreaObj.HDR/.LOD/.TEX (in temp/), the generated
# f16.json (in build/generated_fixtures/), and FALCON4.ct (from the install
# when available, else the bundled fixture).
#
# Phase 0c (FLIGHT_CONTROL_NEXT_STEPS.md §3.2): the takeoff_only and
# landing_only scenarios are isolated diagnostic scenarios. Each emits a
# per-tick CSV trace via fcs_trace_path for offline plotting.
if(F4_INSTALL AND EXISTS "${F4_INSTALL}/terrdata/objects/FALCON4.ct")
    set(F4_DIGI_CLASS_TABLE "${F4_INSTALL}/terrdata/objects/FALCON4.ct")
else()
    set(F4_DIGI_CLASS_TABLE "${CMAKE_SOURCE_DIR}/f4-world-convert/tests/fixtures/FALCON4.ct")
endif()

# Textured-terrain data for the scenario player: when a real install is
# configured, point the scenarios at its korea theater (raw THEATER.L*/
# O* post files + TEXTURE.BIN/texture.zip/FArtILES tile art). Substituted
# into the scenario templates as @F4_SCENARIO_THEATER_DIR@; empty (the
# default without F4_INSTALL) degrades to the untextured terrain path.
set(F4_SCENARIO_THEATER_DIR "")
if(F4_INSTALL AND EXISTS "${F4_INSTALL}/terrdata/korea/texture/TEXTURE.BIN")
    set(F4_SCENARIO_THEATER_DIR "${F4_INSTALL}/terrdata/korea")
endif()

# ============================================================================
# Convenience targets — one-command world generation
# ============================================================================
#
#   cmake --build build --target world
#       Converts the bundled save1.cam to JSON, and the bundled THEATER.*
#       files to terrain.json. Both outputs land in build/.
#
#   cmake --build build --target json
#       Just the .cam -> JSON step (no terrain).
#
#   For interactive inspection: run f4-world-viewer, then use File > Open
#   to load the world JSON and terrain JSON files in the build/ directory.
#

# Locate the bundled campaign fixture and terrain data.
set(F4_CAM_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/f4-world-convert/tests/fixtures/save1.cam")
if(NOT EXISTS "${F4_CAM_FIXTURE}")
    set(F4_CAM_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/temp/save1.cam")
endif()
set(F4_TERRAIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/temp")
if(NOT EXISTS "${F4_TERRAIN_DIR}/THEATER.MEA")
    set(F4_TERRAIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/f4-terrain/tests/fixtures")
endif()

# ── Real Falcon install (optional) ─────────────────────────────────────────
# -DF4_INSTALL="D:/SteamLibrary/steamapps/common/Falcon 4.0" enables
# regeneration of korea_real.world.json from the install's save1.cam +
# the REAL theater object DB (FALCON4.PHD/PD -> true airbase ground
# layouts: runways 02/20, taxi polylines, dims). Without it the bundled
# fixture theater DB is used (highway-strip layouts — same structure).
set(F4_INSTALL "" CACHE PATH "Path to a Falcon 4.0 install (for real campaign/theater data)")
if(F4_INSTALL AND EXISTS "${F4_INSTALL}/terrdata/objects")
    set(F4_REAL_WORLD_JSON "${CMAKE_BINARY_DIR}/korea_real.world.json")
    add_custom_target(korea-real-world-json
        COMMAND $<TARGET_FILE:cam2json>
                "${F4_INSTALL}/campaign/SAVE/save1.cam"
                "${F4_REAL_WORLD_JSON}"
                --theater-data "${F4_INSTALL}/terrdata/objects"
        DEPENDS cam2json
        COMMENT "Converting install save1.cam + real theater DB -> korea_real.world.json"
        VERBATIM
    )
endif()

# .cam -> JSON
if(TARGET cam2json)
add_custom_target(json
    COMMAND $<TARGET_FILE:cam2json>
            "${F4_CAM_FIXTURE}"
            "${CMAKE_BINARY_DIR}/save1.world.json"
    DEPENDS cam2json
    COMMENT "Converting save1.cam -> save1.world.json"
    VERBATIM
)

# THEATER.* -> terrain.json
add_custom_target(terrain-json
    COMMAND $<TARGET_FILE:terrain2json>
            "${F4_TERRAIN_DIR}"
            "${CMAKE_BINARY_DIR}/korea.terrain.json"
    DEPENDS terrain2json
    COMMENT "Converting THEATER.* -> korea.terrain.json"
    VERBATIM
)

# Both at once
add_custom_target(world
    COMMAND $<TARGET_FILE:cam2json>
            "${F4_CAM_FIXTURE}"
            "${CMAKE_BINARY_DIR}/save1.world.json"
    COMMAND $<TARGET_FILE:terrain2json>
            "${F4_TERRAIN_DIR}"
            "${CMAKE_BINARY_DIR}/korea.terrain.json"
    DEPENDS cam2json terrain2json
    COMMENT "Generating world JSON + terrain JSON"
    VERBATIM
)
endif()
